{ ExportBarras5min
  Escribe cada barra de la grafica en un archivo CSV (fecha EasyLanguage y hora de CIERRE de la barra).
  Uso: grafica de @ES a 5 minutos, horario extendido, del 1-ago-2026 a hoy. El archivo queda en la carpeta
  de Dropbox de esta computadora. Si el indicador se recalcula, el archivo puede traer lineas repetidas:
  el programa de Python las limpia. NO PROBADO por Claude en TradeStation. }

inputs:
	FilePath( "C:\\Users\\Macbook\\Dropbox\\Bot Claude\\ES_5min.csv" );

variables:
	Linea( "" );

once
begin
	FileAppend( FilePath, "fecha_el,hora_cierre,open,high,low,close,volume" + NewLine );
end;

Linea = NumToStr( Date, 0 ) + "," + NumToStr( Time, 0 ) + "," +
	NumToStr( Open, 2 ) + "," + NumToStr( High, 2 ) + "," +
	NumToStr( Low, 2 ) + "," + NumToStr( Close, 2 ) + "," +
	NumToStr( Volume, 0 ) + NewLine;

FileAppend( FilePath, Linea );
